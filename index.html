            1. Optimal Page Replacement
          

          

            
            
          

        

        
#include <stdio.h>

#define MAX_FRAMES 10
#define MAX_PAGES 50

// Check if page is already in frames (hit)
int isHit(int frames[], int n, int page) {
    for (int i = 0; i < n; i++)
        if (frames[i] == page) return 1;
    return 0;
}

// Find which frame to replace (used farthest in future)
int findOptimal(int frames[], int n, int pages[], int total, int current) {
    int farthest = -1, replaceIdx = 0;

    for (int i = 0; i < n; i++) {
        int j;
        for (j = current + 1; j < total; j++) {
            if (frames[i] == pages[j]) {
                if (j > farthest) {
                    farthest = j;
                    replaceIdx = i;
                }
                break;
            }
        }
        // Page not used in future at all -> best to replace
        if (j == total) return i;
    }
    return replaceIdx;
}

int main() {
    int frames[MAX_FRAMES], pages[MAX_PAGES];
    int n, total, faults = 0, filled = 0;

    printf("Enter number of frames: ");
    scanf("%d", &n);

    printf("Enter number of pages: ");
    scanf("%d", &total);

    printf("Enter page reference string: ");
    for (int i = 0; i < total; i++)
        scanf("%d", &pages[i]);

    // Initialize frames to -1 (empty)
    for (int i = 0; i < n; i++)
        frames[i] = -1;

    printf("\nPage\tFrames\t\t\tStatus\n");
    printf("----\t------\t\t\t------\n");

    for (int i = 0; i < total; i++) {
        printf("%d\t", pages[i]);

        if (isHit(frames, n, pages[i])) {
            // Page Hit
            for (int j = 0; j < n; j++)
                printf("%d ", frames[j]);
            printf("\t\t\tHIT\n");

        } else {
            // Page Fault
            faults++;

            if (filled < n) {
                // Frames not full yet
                frames[filled++] = pages[i];
            } else {
                // Replace optimal page
                int idx = findOptimal(frames, n, pages, total, i);
                frames[idx] = pages[i];
            }

            for (int j = 0; j < n; j++)
                printf("%d ", frames[j]);
            printf("\t\t\tFAULT\n");
        }
    }

    printf("\nTotal Page Faults: %d\n", faults);
    printf("Total Page Hits  : %d\n", total - faults);
    return 0;
}

#include <stdio.h>

int main() {
    int p, r, i, j, count = 0;
    int alloc[10][10], max[10][10], need[10][10];
    int avail[10], finish[10] = {0}, safeSeq[10];

    printf("Enter processes and resources: ");
    scanf("%d %d", &p, &r);

    printf("Enter Allocation matrix:\n");
    for (i = 0; i < p; i++)
        for (j = 0; j < r; j++)
            scanf("%d", &alloc[i][j]);

    printf("Enter Max matrix:\n");
    for (i = 0; i < p; i++)
        for (j = 0; j < r; j++)
            scanf("%d", &max[i][j]);

    printf("Enter Available resources:\n");
    for (i = 0; i < r; i++)
        scanf("%d", &avail[i]);

    // Need = Max - Alloc
    for (i = 0; i < p; i++)
        for (j = 0; j < r; j++)
            need[i][j] = max[i][j] - alloc[i][j];

    // Safety check
    while (count < p) {
        int found = 0;
        for (i = 0; i < p; i++) {
            if (!finish[i]) {
                int ok = 1;
                for (j = 0; j < r; j++)
                    if (need[i][j] > avail[j]) { ok = 0; break; }

                if (ok) {
                    for (j = 0; j < r; j++)
                        avail[j] += alloc[i][j];
                    safeSeq[count++] = i;
                    finish[i] = 1;
                    found = 1;
                }
            }
        }
        if (!found) { printf("Unsafe state!\n"); return 0; }
    }

    printf("Safe sequence: ");
    for (i = 0; i < p; i++)
        printf("P%d ", safeSeq[i]);
    printf("\n");
    return 0;
}

            4. C-SCAN Disk Scheduling
          

          

            
            
          

        

        
#include <stdio.h>
#include <stdlib.h>

int main() {
    int t[25], n, h, total, i, j, temp, pos, k = 0, sum = 0;
    int atr[25];

    printf("Enter number of requests: ");  scanf("%d", &n);
    printf("Enter head position: ");       scanf("%d", &h);
    printf("Enter total tracks: ");        scanf("%d", &total);

    printf("Enter track numbers:\n");
    for (i = 0; i < n; i++) scanf("%d", &t[i]);

    // Add head, 0, and max track to array
    t[n] = h; t[n+1] = 0; t[n+2] = total - 1;
    n += 3;

    // Sort
    for (i = 0; i < n-1; i++)
        for (j = 0; j < n-i-1; j++)
            if (t[j] > t[j+1]) { temp = t[j]; t[j] = t[j+1]; t[j+1] = temp; }

    // Find head position
    for (i = 0; i < n; i++)
        if (t[i] == h) { pos = i; break; }

    // C-SCAN: go right to end, jump to 0, come back to head
    for (i = pos; i < n; i++) atr[k++] = t[i];
    for (i = 0; i < pos; i++) atr[k++] = t[i];

    // Calculate total movement
    for (i = 0; i < k-1; i++)
        sum += abs(atr[i+1] - atr[i]);

    printf("\nOrder: ");
    for (i = 0; i < k; i++) printf("%d ", atr[i]);
    printf("\nTotal movement: %d\n", sum);
    return 0;
}

#include <stdio.h>

void leftShift(int *A, int *Q, int n) {
    *A = ((*A << 1) | ((*Q >> (n-1)) & 1)) & ((1 << n) - 1);
    *Q = (*Q << 1) & ((1 << n) - 1);
}

int main() {
    int A = 0, Q, M, n, i;

    printf("Enter Dividend: ");  scanf("%d", &Q);
    printf("Enter Divisor: ");   scanf("%d", &M);
    printf("Enter bits: ");      scanf("%d", &n);

    printf("\nA=%d Q=%d M=%d\n", A, Q, M);

    for (i = 0; i < n; i++) {
        leftShift(&A, &Q, n);

        if ((A >> (n-1)) == 0)  // A positive
            A = A - M;
        else                     // A negative
            A = A + M;

        if ((A >> (n-1)) == 0)  // A positive -> Q0 = 1
            Q = Q | 1;
        else                     // A negative -> Q0 = 0
            Q = Q & (~1);

        printf("Step %d: A=%d Q=%d\n", i+1, A, Q);
    }

    // Final correction
    if ((A >> (n-1)) == 1) {
        A = A + M;
        printf("Final correction: A=%d\n", A);
    }

    printf("\nQuotient = %d\n", Q);
    printf("Remainder = %d\n", A);
    return 0;
}

import java.util.*;

class Hamming {
    public static void main(String[] a) {
        Scanner s=new Scanner(System.in);

        System.out.print("Enter two binary strings: ");
        String x=s.next(),y=s.next();
        int d=0;
        for(int i=0;i<x.length();i++)
            if(x.charAt(i)!=y.charAt(i)) d++;
        System.out.println("Hamming Distance: "+d);

        int[] h=new int[7];
        System.out.print("Enter 4 data bits: ");
        h[2]=s.nextInt(); h[4]=s.nextInt();
        h[5]=s.nextInt(); h[6]=s.nextInt();

        h[0]=h[2]^h[4]^h[6];
        h[1]=h[2]^h[5]^h[6];
        h[3]=h[4]^h[5]^h[6];

        System.out.print("Hamming Code: ");
        for(int i:h) System.out.print(i+" ");

        System.out.println("\nEnter received code: ");
        for(int i=0;i<7;i++) h[i]=s.nextInt();

        int e=(h[0]^h[2]^h[4]^h[6])
            +(h[1]^h[2]^h[5]^h[6])*2
            +(h[3]^h[4]^h[5]^h[6])*4;

        if(e==0) System.out.println("No Error");
        else {
            System.out.println("Error at position: "+e);
            h[e-1]^=1;
            System.out.print("Corrected Code: ");
            for(int i:h) System.out.print(i+" ");
        }
    }
}

#include <stdio.h>

int A, M, Q, Q_1;

void arithmetic_right_shift(int n) {
    int sign = A & (1 << (n - 1));
    int q0 = Q & 1;

    Q = Q >> 1;

    if (A & 1)
        Q = Q | (1 << (n - 1));

    A = A >> 1;

    if (sign)
        A = A | (~((1 << (n - 1)) - 1));

    Q_1 = q0;
}

void booth(int multiplicand, int multiplier, int n) {
    int i;
    int mask = (1 << n) - 1;

    A = 0;
    M = multiplicand & mask;
    Q = multiplier & mask;
    Q_1 = 0;

    printf("\nInitial Values:");
    printf("\nA = %d", A);
    printf("\nM = %d", multiplicand);
    printf("\nQ = %d", multiplier);
    printf("\nQ-1 = %d\n", Q_1);

    for (i = 1; i <= n; i++) {

        if ((Q & 1) == 1 && Q_1 == 0) {
            A = A - M;
            A = A & mask;
            printf("\nStep %d: A = A - M", i);
        }
        else if ((Q & 1) == 0 && Q_1 == 1) {
            A = A + M;
            A = A & mask;
            printf("\nStep %d: A = A + M", i);
        }
        else {
            printf("\nStep %d: No operation", i);
        }

        arithmetic_right_shift(n);

        A = A & mask;
        Q = Q & mask;

        printf("\nA = ");
        printf("%0*d", n, A);
        printf("  Q = ");
        printf("%0*d", n, Q);
        printf("  Q-1 = %d", Q_1);
    }

    {
        long long product;

        product = ((long long)A << n) | Q;

        if (A & (1 << (n - 1))) {
            product = product - ((long long)1 << (2 * n));
        }

        printf("\n\nProduct = %lld\n", product);
    }
}

int main() {
    int multiplicand, multiplier, n;

    printf("Enter multiplicand: ");
    scanf("%d", &multiplicand);

    printf("Enter multiplier: ");
    scanf("%d", &multiplier);

    printf("Enter number of bits: ");
    scanf("%d", &n);

    if (n <= 0 || n >= 16) {
        printf("Please enter number of bits between 1 and 15.\n");
        return 1;
    }

    booth(multiplicand, multiplier, n);

    return 0;
}

            2. SJF (Non-Preemptive)
          

          

            
            
          

        

        
#include <stdio.h>

int main() {
    int n, i, j;
    int burst[20], p[20], wait[20], turnaround[20];
    int total_wait = 0, total_turnaround = 0;

    printf("Enter number of processes: ");
    scanf("%d", &n);

    for (i = 0; i < n; i++) {
        printf("Enter burst time for process %d: ", i + 1);
        scanf("%d", &burst[i]);
        p[i] = i + 1;
    }

    // Sort based on burst time
    for (i = 0; i < n - 1; i++) {
        for (j = 0; j < n - i - 1; j++) {

            if (burst[j] > burst[j + 1]) {

                int temp = burst[j];
                burst[j] = burst[j + 1];
                burst[j + 1] = temp;

                temp = p[j];
                p[j] = p[j + 1];
                p[j + 1] = temp;
            }
        }
    }

    wait[0] = 0;

    for (i = 1; i < n; i++) {
        wait[i] = 0;

        for (j = 0; j < i; j++)
            wait[i] += burst[j];

        total_wait += wait[i];
    }

    printf("\nProcess\tBurst Time\tWaiting Time\tTurnaround Time\n");

    for (i = 0; i < n; i++) {

        turnaround[i] = burst[i] + wait[i];
        total_turnaround += turnaround[i];

        printf("P%d\t%d\t\t%d\t\t%d\n",
               p[i], burst[i], wait[i], turnaround[i]);
    }

    printf("\nAverage Waiting Time = %.2f",
           (float)total_wait / n);

    printf("\nAverage Turnaround Time = %.2f\n",
           (float)total_turnaround / n);

    return 0;
}

#include <stdio.h>

int main() {
    int n, rss, fa[20], rsa[50];
    int i, j, cp = 0, pfc = 0, npf;

    // Number of frames
    printf("Enter number of frames: ");
    scanf("%d", &n);

    // Reference string size
    printf("Enter number of pages in reference string: ");
    scanf("%d", &rss);

    // Reference string
    printf("Enter Reference String:\n");

    for (i = 0; i < rss; i++) {
        scanf("%d", &rsa[i]);
    }

    // Initialize frames
    for (i = 0; i < n; i++) {
        fa[i] = -1;
    }

    printf("\nCURRENT_PAGE\tPAGES_IN_FRAME\t\tPAGE_FAULT_OCCURRED?\n");

    // FIFO algorithm
    for (i = 0; i < rss; i++) {

        printf("\n\t%d\t\t", rsa[i]);
        npf = 0;

        // Check if page already exists
        for (j = 0; j < n; j++) {

            if (fa[j] == rsa[i]) {
                npf = 1;
                printf("\t\t\tNO");
                break;
            }
        }

        // Page fault
        if (npf == 0) {

            pfc++;

            fa[cp] = rsa[i];

            // Move to next frame
            cp = (cp + 1) % n;

            // Display frames
            for (j = 0; j < n; j++) {

                if (fa[j] != -1)
                    printf("%d\t", fa[j]);
                else
                    printf("-\t");
            }

            printf("\tYES");
        }
    }

    printf("\n\nTotal number of page faults: %d\n", pfc);

    return 0;
}

#include <stdio.h>
#include <conio.h>

void main() {

    int ms, ps, nop, np, rempages;
    int i, j, x, y, pa, offset;

    int s[10], fno[10][20];

    clrscr();

    // Memory size
    printf("\nEnter the memory size -- ");
    scanf("%d", &ms);

    // Page size
    printf("\nEnter the page size -- ");
    scanf("%d", &ps);

    // Number of pages
    nop = ms / ps;

    printf("\nThe no. of pages available in memory are -- %d", nop);

    // Number of processes
    printf("\nEnter number of processes -- ");
    scanf("%d", &np);

    rempages = nop;

    // Page table for each process
    for (i = 1; i <= np; i++) {

        printf("\nEnter no. of pages required for p[%d]-- ", i);
        scanf("%d", &s[i]);

        if (s[i] > rempages) {
            printf("\nMemory is Full");
            break;
        }

        rempages = rempages - s[i];

        printf("\nEnter page table for p[%d] --- ", i);

        for (j = 0; j < s[i]; j++) {
            scanf("%d", &fno[i][j]);
        }
    }

    // Logical to physical address
    printf("\nEnter Logical Address to find Physical Address");

    printf("\nEnter process no., page number, and offset -- ");

    scanf("%d %d %d", &x, &y, &offset);

    // Validation
    if (x > np || y >= s[x] || offset >= ps) {

        printf("\nInvalid Process or Page Number or Offset");

    }
    else {

        pa = fno[x][y] * ps + offset;

        printf("\nThe Physical Address is -- %d", pa);
    }

    getch();
}

#include <stdio.h>
#include <stdlib.h>

int main() {

    int queue[100], visited[100];
    int n, i, j, head;
    int pos, min, diff;
    int seek = 0;
    float avg;

    printf("*** SSTF Disk Scheduling Algorithm ***\n");

    printf("Enter the size of the queue: ");
    scanf("%d", &n);

    printf("Enter the queue (track numbers):\n");

    for (i = 0; i < n; i++) {
        scanf("%d", &queue[i]);
        visited[i] = 0;
    }

    printf("Enter the initial head position: ");
    scanf("%d", &head);

    printf("\nOrder of head movement:\n");

    for (i = 0; i < n; i++) {

        min = 100000;
        pos = -1;

        // Find closest unvisited request
        for (j = 0; j < n; j++) {

            if (!visited[j]) {

                diff = abs(queue[j] - head);

                if (diff < min) {
                    min = diff;
                    pos = j;
                }
            }
        }

        // Process closest request
        visited[pos] = 1;

        seek += abs(queue[pos] - head);

        head = queue[pos];

        printf("%d ", head);
    }

    avg = (float)seek / n;

    printf("\n\nTotal Seek Time = %d", seek);
    printf("\nAverage Seek Time = %.2f\n", avg);

    return 0;
}
